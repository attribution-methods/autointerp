     1	"""Canonical metric registry — the only legitimate path to a MetricResult.
     2	
     3	The investigation pipeline forbids agents from writing ``MetricResult``
     4	artifacts directly. They must call ``compute_metric``, which:
     5	
     6	1. dispatches to the canonical implementation for the named ``MetricName``;
     7	2. validates the input shape that implementation declares;
     8	3. computes the value, range-checked against ``METRIC_META`` for that metric;
     9	4. builds a ``MetricResult`` payload (id, value, threshold, pass/fail,
    10	   metadata including unclipped value and inputs_hash);
    11	5. registers a one-time provenance token in ``state.pending_provenance_tokens``
    12	   so ``commit_artifact("MetricResult", ...)`` can verify the agent did not
    13	   tamper with the value before committing.
    14	
    15	If an agent wants a metric that isn't here, the right path is a spec revision
    16	(extending ``MetricName`` is a PR-time change, not a runtime one).
    17	"""

    52	def _require_keys(name: MetricName, inputs: dict[str, Any], keys: Iterable[str]) -> None:
    53	    missing = [k for k in keys if k not in inputs]
    54	    if missing:
    55	        raise MetricRegistryError(
    56	            f"{name.value}: missing required input keys: {sorted(missing)}; "
    57	            f"got {sorted(inputs)}"
    58	        )

    61	def _as_float_list(name: MetricName, key: str, raw: Any) -> list[float]:
    62	    if not isinstance(raw, list) or not raw:
    63	        raise MetricRegistryError(f"{name.value}.{key} must be a non-empty list")
    64	    out: list[float] = []
    65	    for i, v in enumerate(raw):
    66	        try:
    67	            out.append(float(v))
    68	        except (TypeError, ValueError) as exc:
    69	            raise MetricRegistryError(
    70	                f"{name.value}.{key}[{i}] is not a number: {v!r}"
    71	            ) from exc
    72	    return out

    75	def _accuracy(inputs: dict[str, Any]) -> float:
    76	    """Mean(predictions[i] == labels[i]).
    77	
    78	    Required inputs:
    79	      - predictions: list[Any] of model outputs
    80	      - labels: list[Any] of gold labels (equal length to predictions)
    81	    """
    82	    _require_keys(MetricName.ACCURACY, inputs, ("predictions", "labels"))
    83	    preds = inputs["predictions"]
    84	    labels = inputs["labels"]
    85	    if not isinstance(preds, list) or not isinstance(labels, list):
    86	        raise MetricRegistryError("accuracy: predictions and labels must be lists")
    87	    if len(preds) != len(labels):
    88	        raise MetricRegistryError(
    89	            f"accuracy: predictions ({len(preds)}) and labels ({len(labels)}) "
    90	            "have different lengths"
    91	        )
    92	    if not preds:
    93	        raise MetricRegistryError("accuracy: predictions list is empty")
    94	    correct = sum(1 for p, l in zip(preds, labels) if p == l)
    95	    return correct / len(preds)
