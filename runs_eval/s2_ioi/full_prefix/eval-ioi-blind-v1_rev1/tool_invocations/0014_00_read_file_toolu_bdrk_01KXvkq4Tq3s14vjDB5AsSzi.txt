    31	def resolve_model_id(model_name: str) -> str:
    32	    return MODEL_ALIASES.get(model_name, model_name)
    33	
    34	
    35	def infer_model_type(model_name: str) -> str:
    36	    lower = model_name.lower()
    37	    if "llama" in lower:
    38	        return "llama"
    39	    if "qwen" in lower:
    40	        return "qwen"
    41	    if "gemma" in lower:
    42	        return "gemma"
    43	    if "mistral" in lower:
    44	        return "mistral"
    45	    if "deepseek" in lower:
    46	        return "deepseek"
    47	    return "unknown"
    48	
    49	
    50	def make_quantization_config(mode: Optional[str]) -> Optional[BitsAndBytesConfig]:
    51	    if mode is None:
    52	        return None
    53	    if mode == "8bit":
    54	        return BitsAndBytesConfig(load_in_8bit=True)
    55	    if mode == "4bit":
    56	        return BitsAndBytesConfig(load_in_4bit=True)
    57	    raise ValueError(f"Unknown quantization mode: {mode}")

    59	
    60	@dataclass
    61	class ModelHandle:
    62	    model_name: str
    63	    model_id: str
    64	    model: Any
    65	    tokenizer: Any
    66	    device: str = "cuda"
    67	    dtype: torch.dtype = torch.float32
    68	    model_type: str = "unknown"
    69	
    70	    @property
    71	    def n_layers(self) -> int:
    72	        if hasattr(self.model, "model"):
    73	            inner = self.model.model
    74	            if hasattr(inner, "language_model") and hasattr(inner.language_model, "layers"):
    75	                return len(inner.language_model.layers)
    76	            if hasattr(inner, "layers"):
    77	                return len(inner.layers)
    78	            if hasattr(inner, "model") and hasattr(inner.model, "layers"):
    79	                return len(inner.model.layers)
    80	        if hasattr(self.model, "transformer") and hasattr(self.model.transformer, "h"):
    81	            return len(self.model.transformer.h)
    82	        config = self.model.config
    83	        for name in ("num_hidden_layers", "n_layer", "num_layers"):
    84	            if hasattr(config, name):
    85	                return int(getattr(config, name))
    86	        if hasattr(config, "text_config") and hasattr(config.text_config, "num_hidden_layers"):
    87	            return int(config.text_config.num_hidden_layers)
    88	        raise ValueError("Could not infer number of layers")
    89	
    90	    @property
    91	    def d_model(self) -> int:
    92	        config = self.model.config
    93	        for name in ("hidden_size", "d_model", "dim", "n_embd"):
    94	            if hasattr(config, name):
    95	                return int(getattr(config, name))
    96	        if hasattr(config, "text_config") and hasattr(config.text_config, "hidden_size"):
    97	            return int(config.text_config.hidden_size)
    98	        raise ValueError("Could not infer hidden size")
    99	
   100	    @property
   101	    def n_heads(self) -> int:
   102	        config = self.model.config
   103	        for name in ("num_attention_heads", "n_head", "num_heads"):
   104	            if hasattr(config, name):
   105	                return int(getattr(config, name))
   106	        if hasattr(config, "text_config") and hasattr(config.text_config, "num_attention_heads"):
   107	            return int(config.text_config.num_attention_heads)
   108	        raise ValueError("Could not infer number of attention heads")
   109	
   110	    def input_device(self) -> torch.device:
   111	        return next(self.model.parameters()).device
   112	
   113	    def layer(self, layer_idx: int) -> Any:
   114	        if layer_idx < 0:
   115	            layer_idx = self.n_layers + layer_idx
   116	        if hasattr(self.model, "model"):
   117	            inner = self.model.model
   118	            if hasattr(inner, "language_model") and hasattr(inner.language_model, "layers"):
   119	                return inner.language_model.layers[layer_idx]
   120	            if hasattr(inner, "layers"):
   121	                return inner.layers[layer_idx]
   122	            if hasattr(inner, "model") and hasattr(inner.model, "layers"):
   123	                return inner.model.layers[layer_idx]
   124	        if hasattr(self.model, "transformer") and hasattr(self.model.transformer, "h"):
   125	            return self.model.transformer.h[layer_idx]
   126	        raise ValueError(f"Could not access layer {layer_idx}")
   127	
   128	    def filter_messages(self, messages: List[Dict[str, str]]) -> List[Dict[str, str]]:
   129	        if self.model_type in MODELS_WITHOUT_SYSTEM_ROLE:
   130	            return [m for m in messages if m.get("role") != "system"]
   131	        return messages
   132	
   133	    def format_messages(
   134	        self,
   135	        messages: List[Dict[str, str]],
   136	        add_generation_prompt: bool = True,
   137	    ) -> str:
   138	        messages = self.filter_messages(messages)
   139	        if hasattr(self.tokenizer, "apply_chat_template"):
   140	            return self.tokenizer.apply_chat_template(
   141	                messages,
   142	                tokenize=False,
   143	                add_generation_prompt=add_generation_prompt,
   144	            )
   145	        parts = []
   146	        for message in messages:
   147	            role = message.get("role", "user").capitalize()
   148	            content = message.get("content", "")
   149	            if content:
   150	                parts.append(f"{role}: {content}")
   151	        if add_generation_prompt:
   152	            parts.append("Assistant:")
   153	        return "\n\n".join(parts)
   154	
   155	    def tokenize(self, text: str, padding: bool = False) -> Dict[str, torch.Tensor]:
   156	        return self.tokenizer(
   157	            text,
   158	            return_tensors="pt",
   159	            padding=padding,
   160	            add_special_tokens=False,
   161	        ).to(self.input_device())
   162	
   163	    def generate(
   164	        self,
   165	        prompt: str,
   166	        max_new_tokens: int = 128,
   167	        temperature: float = 0.0,
   168	        **generation_kwargs: Any,
   169	    ) -> str:
   170	        inputs = self.tokenize(prompt)
   171	        input_length = inputs["input_ids"].shape[1]
   172	        kwargs = {
   173	            "max_new_tokens": max_new_tokens,
   174	            "pad_token_id": self.tokenizer.pad_token_id,
   175	        }
   176	        if temperature > 0:
   177	            kwargs.update({"do_sample": True, "temperature": temperature})
   178	            kwargs.update(generation_kwargs)
   179	        with torch.no_grad():
   180	            output_ids = self.model.generate(**inputs, **kwargs)
   181	        new_tokens = output_ids[0][input_length:]
   182	        return self.tokenizer.decode(new_tokens, skip_special_tokens=True).strip()
   183	
   184	    def generate_batch(
   185	        self,
   186	        prompts: List[str],
   187	        max_new_tokens: int = 128,
   188	        temperature: float = 0.0,
   189	        **generation_kwargs: Any,
   190	    ) -> List[str]:
   191	        inputs = self.tokenizer(
   192	            prompts,
   193	            return_tensors="pt",
   194	            padding=True,
   195	            add_special_tokens=False,
   196	        ).to(self.input_device())
   197	        input_lengths = inputs["attention_mask"].sum(dim=1).tolist()
   198	        kwargs = {
   199	            "max_new_tokens": max_new_tokens,
   200	            "pad_token_id": self.tokenizer.pad_token_id,
   201	        }
   202	        if temperature > 0:
   203	            kwargs.update({"do_sample": True, "temperature": temperature})
   204	            kwargs.update(generation_kwargs)
   205	        with torch.no_grad():
   206	            output_ids = self.model.generate(**inputs, **kwargs)
   207	        outputs = []
   208	        for row, input_length in enumerate(input_lengths):
   209	            new_tokens = output_ids[row][input_length:]
   210	            outputs.append(self.tokenizer.decode(new_tokens, skip_special_tokens=True).strip())
   211	        return outputs
   212	
   213	    def cleanup(self) -> None:
   214	        if hasattr(self.model, "cpu"):
   215	            self.model.cpu()
   216	        torch.cuda.empty_cache()
   217	
   218	
   219	def load_model(
   220	    model_name: str,
   221	    device: str = "cuda",
   222	    dtype: torch.dtype = torch.float32,
   223	    quantization: Optional[str] = None,
   224	    trust_remote_code: bool = True,
   225	) -> ModelHandle:
   226	    model_id = resolve_model_id(model_name)
   227	    tokenizer = AutoTokenizer.from_pretrained(model_id, trust_remote_code=trust_remote_code)
   228	    tokenizer.padding_side = "left"
   229	    if tokenizer.pad_token is None:
   230	        tokenizer.pad_token = tokenizer.eos_token
   231	
   232	    quantization_config = make_quantization_config(quantization)
   233	    load_kwargs: Dict[str, Any] = {
   234	        "pretrained_model_name_or_path": model_id,
   235	        "trust_remote_code": trust_remote_code,
   236	        "device_map": "auto" if device == "cuda" else None,
   237	        "attn_implementation": "eager",
   238	    }
   239	    if quantization_config is not None:
   240	        load_kwargs["quantization_config"] = quantization_config
   241	    elif model_name not in PREQUANTIZED_ALIASES:
   242	        load_kwargs["dtype"] = dtype
   243	
   244	    try:
   245	        model = AutoModelForCausalLM.from_pretrained(**load_kwargs)
   246	    except TypeError:
   247	        if "dtype" in load_kwargs:
   248	            load_kwargs["torch_dtype"] = load_kwargs.pop("dtype")
   249	        model = AutoModelForCausalLM.from_pretrained(**load_kwargs)
   250	    if device != "cuda":
   251	        model = model.to(device)
   252	    model.eval()
   253	    return ModelHandle(
   254	        model_name=model_name,
   255	        model_id=model_id,
   256	        model=model,
   257	        tokenizer=tokenizer,
   258	        device=device,
   259	        dtype=dtype,
   260	        model_type=infer_model_type(model_name),
   261	    )
